    <script>
        // 9명 멤버 명단 및 고정 비밀번호(PIN), 마니또 매칭 데이터
        const members = [
            { name: "김여원", pin: "1001", target: "손희정" },
            { name: "손희정", pin: "1002", target: "공현희" },
            { name: "공현희", pin: "1003", target: "강한결" },
            { name: "강한결", pin: "1004", target: "문소윤" },
            { name: "문소윤", pin: "1005", target: "이윤서" },
            { name: "이윤서", pin: "1006", target: "신가연" },
            { name: "신가연", pin: "1007", target: "전효원" },
            { name: "전효원", pin: "1008", target: "한은지" },
            { name: "한은지", pin: "1009", target: "김여원" }
        ];

        // 페이지 로드 시 초기화 및 로컬 스토리지 데이터 복구
        window.onload = function() {
            const select = document.getElementById("myName");
            const grid = document.getElementById("memberGrid");

            members.forEach(m => {
                let opt = document.createElement("option");
                opt.value = m.name;
                opt.textContent = m.name;
                select.appendChild(opt);

                let badge = document.createElement("div");
                badge.className = "member-badge";
                badge.textContent = m.name;
                grid.appendChild(badge);
            });

            // 저장된 마지막 선택 이름 불러오기 (선택사항 편의성)
            const savedName = localStorage.getItem("manitto_last_name");
            if (savedName) {
                select.value = savedName;
            }

            loadMissions();
        };

        // 탭 전환 기능
        function switchTab(tabId) {
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));

            if (tabId === 'check') {
                document.querySelector('.tabs button:nth-child(1)').classList.add('active');
                document.getElementById('tab-check').classList.add('active');
            } else if (tabId === 'members') {
                document.querySelector('.tabs button:nth-child(2)').classList.add('active');
                document.getElementById('tab-members').classList.add('active');
            } else if (tabId === 'missions') {
                document.querySelector('.tabs button:nth-child(3)').classList.add('active');
                document.getElementById('tab-missions').classList.add('active');
            }
        }

        // 비밀번호 확인 후 마니또 조회 (localStorage로 확인 상태 저장 가능)
        function checkManitto() {
            const name = document.getElementById("myName").value;
            const pin = document.getElementById("myPin").value;
            const errorDiv = document.getElementById("pinError");
            const resultBox = document.getElementById("resultBox");

            errorDiv.textContent = "";
            resultBox.style.display = "none";

            if (!name) {
                errorDiv.textContent = "본인 이름을 선택해주세요.";
                return;
            }
            if (!pin) {
                errorDiv.textContent = "비밀번호 4자리를 입력해주세요.";
                return;
            }

            const member = members.find(m => m.name === name);
            if (member && member.pin === pin) {
                localStorage.setItem("manitto_last_name", name); // 마지막 조회 이름 저장
                resultBox.style.display = "block";
                document.getElementById("resultTitle").textContent = `${member.name}님의 마니또는?!`;
                document.getElementById("resultDesc").textContent = `🎁 ${member.target} 님입니다!`;
            } else {
                errorDiv.textContent = "비밀번호가 일치하지 않습니다. 다시 확인해주세요.";
            }
        }

        // 익명 미션 등록 (LocalStorage 연동으로 새로고침해도 데이터 유지)
        function postMission() {
            const category = document.getElementById("missionCategory").value;
            const text = document.getElementById("missionText").value.trim();

            if (!text) {
                alert("내용을 입력해주세요!");
                return;
            }

            const newMission = {
                id: Date.now(),
                category: category,
                text: text,
                date: new Date().toLocaleDateString() + " " + new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})
            };

            let missions = JSON.parse(localStorage.getItem("manitto_missions") || "[]");
            missions.unshift(newMission); // 최신 글이 위로 오도록 추가
            localStorage.setItem("manitto_missions", JSON.stringify(missions));

            document.getElementById("missionText").value = "";
            loadMissions();
        }

        // 익명 미션 불러오기 (localStorage에서 가져옴)
        function loadMissions() {
            const feed = document.getElementById("missionFeed");
            feed.innerHTML = "";

            let missions = JSON.parse(localStorage.getItem("manitto_missions") || "[]");

            if (missions.length === 0) {
                feed.innerHTML = `<p style="text-align: center; color: #b2bec3; padding: 20px;">아직 등록된 익명 미션이 없습니다. 첫 글을 남겨보세요!</p>`;
                return;
            }

            missions.forEach(m => {
                let div = document.createElement("div");
                div.className = "mission-item";
                div.innerHTML = `
                    <div class="mission-header">
                        <span class="tag">${m.category}</span>
                        <span>${m.date}</span>
                    </div>
                    <p>${escapeHtml(m.text)}</p>
                `;
                feed.appendChild(div);
            });
        }

        // XSS 방지 유틸
        function escapeHtml(str) {
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;").replace(/\n/g, "<br>");
        }
